40|def _validate_amount(amount: Any) -> tuple[bool, str]:
41|    """校验金额合法性。返回 (ok, error_code)。"""
42|    if amount is None:
43|        return False, "INVALID_AMOUNT"
44|    # 拒绝 bool（bool 是 int 子类）
45|    if isinstance(amount, bool):
46|        return False, "INVALID_AMOUNT"
47|    try:
48|        val = float(amount)
49|    except (TypeError, ValueError):
50|        return False, "INVALID_AMOUNT"
51|    if val <= 0:
52|        return False, "INVALID_AMOUNT"
53|    if val > 1e9:
54|        return False, "OVERSIZED_AMOUNT"
55|    return True, ""

94|def _bj_week_start(dt_utc: datetime.datetime) -> datetime.datetime:
95|    """给定 UTC 时间，返回北京时间周一 00:00 的 UTC 时间戳。
96|    使用固定 UTC+8 偏移，避免对 tzdata 的依赖。"""
97|    # 先转成北京时间 (UTC+8)
98|    bj = dt_utc + datetime.timedelta(hours=8)
99|    # 本周一 00:00（北京时间）
100|    monday_bj = bj - datetime.timedelta(days=bj.weekday())
101|    monday_bj = monday_bj.replace(hour=0, minute=0, second=0, microsecond=0)
102|    # 再转回 UTC，返回 naive datetime（与旧行为一致）
103|    return (monday_bj - datetime.timedelta(hours=8)).replace(tzinfo=None)
104|

106|def _is_birthday_week(dt_utc: datetime.datetime) -> bool:
107|    """给定 UTC 时间，判断其北京时间所在周是否包含 4月5日或11月15日。
108|    使用固定 UTC+8 偏移，避免对 tzdata 的依赖。"""
109|    bj = dt_utc + datetime.timedelta(hours=8)
110|    year = bj.year
111|    # 本周一（含）到下周日（含）
112|    monday = bj - datetime.timedelta(days=bj.weekday())
113|    monday = monday.replace(hour=0, minute=0, second=0, microsecond=0)
114|    week_start = monday.date()
115|    week_end = week_start + datetime.timedelta(days=6)
116|
117|    apr5 = datetime.date(year, 4, 5)
118|    nov15 = datetime.date(year, 11, 15)
119|
120|    return (week_start <= apr5 <= week_end) or (week_start <= nov15 <= week_end)

136|def _get_supabase():
137|    """延迟获取 server.py 中初始化的 supabase 客户端。"""
138|    try:
139|        import server
140|        return server.supabase
141|    except Exception:
142|        return None

144|
145|def _rpc(name: str, params: dict) -> dict:
146|    """调用 PostgreSQL RPC 函数。"""
147|    sb = _get_supabase()
148|    if sb is None:
149|        return _format_result(False, "数据库未连接", error_code="DB_UNAVAILABLE")
150|    try:
151|        # Supabase Python 客户端通过 .rpc(name, params) 调用函数
152|        resp = sb.rpc(name, params).execute()
153|        # rpc 返回的数据在 resp.data 中
154|        if resp.data is not None:
155|            return resp.data
156|        return _format_result(False, "RPC 返回空数据", error_code="RPC_EMPTY")
157|    except Exception as e:
158|        return _format_result(False, f"RPC 调用失败: {e}", error_code="RPC_ERROR")

199|def _iso_week_label(dt: datetime.datetime) -> str:
200|    """计算 ISO 周标签，格式 YYYYW##（如 2026W33）。
201|    北京时间（UTC+8）为准，与 console.html 端 JS 计算保持一致。"""
202|    bj = dt + datetime.timedelta(hours=8)
203|    iso_year, iso_week, _ = bj.isocalendar()
204|    return f"{iso_year}W{iso_week:02d}"
